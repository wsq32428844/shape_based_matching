    cv::Mat image(height, width, channels == 3 ? CV_8UC3 : CV_8UC1, image_data);
    std::vector<line2Dup::Match> matches_vec;
    matches_vec = static_cast<line2Dup::Detector*>(detector)->match(image, threshold);
